cmake_minimum_required(VERSION 3.20)
project(extio_sdr_bridge LANGUAGES CXX)
add_library(ExtIO_SDR_Bridge SHARED extio_bridge.cpp extio_bridge.def)
target_compile_features(ExtIO_SDR_Bridge PRIVATE cxx_std_17)
target_compile_definitions(ExtIO_SDR_Bridge PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ExtIO_SDR_Bridge PRIVATE Ws2_32)
set_target_properties(ExtIO_SDR_Bridge PROPERTIES OUTPUT_NAME "ExtIO_SDR_Bridge")
add_executable(extio_smoke extio_smoke.cpp)
add_dependencies(extio_smoke ExtIO_SDR_Bridge)
